import os from google import genai from google.genai import types # 1. Khởi tạo client (Tự động lấy API key từ biến môi trường GEMINI_API_KEY) client = genai.Client() # 2. Cấu hình nội dung cuộc trò chuyện prompt = "Chào Gemini! Hãy giới thiệu ngắn gọn về bản thân bạn bằng một câu thật ấn tượng." print(f"🤖 Đang gửi yêu cầu: '{prompt}'\n") # 3. Gọi model Gemini 2.5 Flash (hoặc model phù hợp) response = client.models.generate_content( model="gemini-2.5-flash", contents=prompt, config=types.GenerateContentConfig( temperature=0.7, # Độ sáng tạo (0.0 - 2.0) max_output_tokens=150, # Giới hạn độ dài phản hồi ), ) # 4. Hiển thị kết quả print("✨ Phản hồi từ Gemini:") print("-" * 40) print(response.text) print("-" * 40)
